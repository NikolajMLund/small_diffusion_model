\section{Tanker om fordelingsparametre $\mu$}
\subsection*{Nyttefunktion}

\begin{align}
d_t = \left[ \sum_{a=0}^{A} f\big(\tilde{d}_{a,t}\big) \right]^{\frac{1}{1-\sigma}}
\end{align}

hvor
\begin{align}
f(x) = \mu^\sigma x^{1-\sigma}
\end{align}

\subsection*{Marshalliansk efterspørgsel}

I den mest generelle form:
\begin{align}
d_{f,a,t} 
= \mu_{f,a,t} \left( \frac{p_{f,a,t}}{p_{f,t}} \right)^{-\sigma} d_{f,t} 
+ h_f d_{f,a-1,t-1}
\end{align}

\subsection*{Parametrisering af $\mu$}

Vi har stadig et fuldt sæt af $\mu$'er.

Vi kunne sige:
\begin{align}
\mu_{f,0,t} \text{ varierer over tid}, \quad 
\mu_{f,a} = \mu_{f,a,t} \quad \forall t, \; \forall a > 0
\end{align}

Implikation: elbiler indfases via $\mu_{f,0,t}$, mens alderspræferencer er tidsinvariante.

Så bliver:
\begin{align}
d_{f,a,t} 
= \mu_{f,a} \left( \frac{p_{f,a,t}}{p_{f,t}} \right)^{-\sigma} d_{f,t}
+ h_f d_{f,a-1,t-1}
\end{align}

\paragraph{Problem}
Hvis $d_{f,a-1,t-1} = 0$ og $\mu_{f,a} > 0$, kræver det at priser går mod uendelig for at få $d_{f,a,t} \to 0$.

\subsection*{Alternativ specifikation}

Antag:
\begin{align}
\mu_{f,t} = \mu_{f,a,t} \quad \forall t, \; \forall a > 0
\end{align}

Så bliver:
\begin{align}
d_{f,a,t} 
= \mu_{f,t} \left( \frac{p_{f,a,t}}{p_{f,t}} \right)^{-\sigma} d_{f,t}
+ h_f d_{f,a-1,t-1}
\end{align}

Hvis $d_{f,a-1,t-1} = 0$:
\begin{align}
d_{f,a,t} 
= \mu_{f,t} \left( \frac{p_{f,a,t}}{p_{f,t}} \right)^{-\sigma} d_{f,t}
\end{align}

Dynamik:
\begin{align}
d_{f,a+1,t+1} - h_f d_{f,a,t}
= \mu_{f,t+1} \left( \frac{p_{f,a+1,t+1}}{p_{f,t+1}} \right)^{-\sigma} d_{f,t+1}
\end{align}

Dette giver sandsynligvis den øverste del af en S-kurve (aftagende effekt), hvor:
\begin{align}
\mu_{f,t} \to \mu_f
\end{align}

Den nederste (accelererende) del af S-kurven kan styres via:
\begin{align}
\frac{\partial \mu_{f,t}}{\partial t}
\end{align}
